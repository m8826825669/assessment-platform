CREATE FUNCTION question_identity_guard() RETURNS trigger LANGUAGE plpgsql
        SET search_path = pg_catalog, public AS $$
        BEGIN
            IF TG_OP = 'DELETE' THEN
                RAISE EXCEPTION 'Question identity is retained' USING ERRCODE = '42501';
            END IF;
            IF NEW.id IS DISTINCT FROM OLD.id OR NEW.tenant_id IS DISTINCT FROM OLD.tenant_id
               OR NEW.created_by IS DISTINCT FROM OLD.created_by OR NEW.created_at IS DISTINCT FROM OLD.created_at
               OR NEW.revision::bigint <> OLD.revision::bigint + 1 THEN
                RAISE EXCEPTION 'Only the next revision is allowed' USING ERRCODE = '23514';
            END IF;
            RETURN NEW;
        END $$;

CREATE TRIGGER question_identity_guard BEFORE UPDATE OR DELETE ON questions FOR EACH ROW EXECUTE FUNCTION question_identity_guard();

CREATE FUNCTION question_version_guard() RETURNS trigger LANGUAGE plpgsql
        SET search_path = pg_catalog, public AS $$
        DECLARE current_revision integer;
        BEGIN
            IF TG_OP <> 'INSERT' THEN
                RAISE EXCEPTION 'Saved question versions are immutable' USING ERRCODE = '42501';
            END IF;
            SELECT q.revision INTO current_revision FROM public.questions q
                WHERE q.tenant_id = NEW.tenant_id AND q.id = NEW.question_id FOR UPDATE;
            IF current_revision IS NULL OR current_revision <> NEW.version_number THEN
                RAISE EXCEPTION 'Version must match the current revision' USING ERRCODE = '23514';
            END IF;
            RETURN NEW;
        END $$;

CREATE TRIGGER question_version_guard BEFORE INSERT OR UPDATE OR DELETE ON question_versions FOR EACH ROW EXECUTE FUNCTION question_version_guard();

CREATE FUNCTION question_review_guard() RETURNS trigger LANGUAGE plpgsql
        SET search_path = pg_catalog, public AS $$
        DECLARE q public.questions%ROWTYPE; v public.question_versions%ROWTYPE;
                previous public.question_review_events%ROWTYPE; role_name text; actor uuid;
        BEGIN
            IF TG_OP <> 'INSERT' THEN RAISE EXCEPTION 'Review decisions are immutable' USING ERRCODE='42501'; END IF;
            actor := NULLIF(current_setting('app.actor_user_id', true), '')::uuid;
            PERFORM pg_advisory_xact_lock(hashtextextended(NEW.tenant_id::text || NEW.question_id::text, 0));
            SELECT * INTO q FROM public.questions WHERE tenant_id=NEW.tenant_id AND id=NEW.question_id;
            SELECT * INTO v FROM public.question_versions WHERE tenant_id=NEW.tenant_id AND question_id=NEW.question_id AND id=NEW.version_id;
            IF q.id IS NULL OR v.id IS NULL OR q.created_by IS DISTINCT FROM NEW.originating_author
               OR v.content_hash IS DISTINCT FROM NEW.content_hash OR actor IS DISTINCT FROM NEW.actor_user_id THEN
                RAISE EXCEPTION 'Review binding mismatch' USING ERRCODE='23514';
            END IF;
            SELECT m.role INTO role_name FROM public.memberships m JOIN public.institutions i ON i.id=m.tenant_id
                JOIN public.users u ON u.id=m.user_id WHERE m.tenant_id=NEW.tenant_id AND m.user_id=actor
                AND m.status='active' AND i.status='active' AND u.is_active;
            SELECT * INTO previous FROM public.question_review_events WHERE tenant_id=NEW.tenant_id AND version_id=NEW.version_id ORDER BY sequence DESC LIMIT 1;
            IF NEW.expected_revision <> q.revision OR NEW.sequence <> COALESCE(previous.sequence,0)+1 THEN
                RAISE EXCEPTION 'Stale review' USING ERRCODE='23514';
            END IF;
            IF NEW.action <> 'retire' AND v.version_number <> q.revision THEN
                RAISE EXCEPTION 'Only the current version may be reviewed' USING ERRCODE='23514';
            END IF;
            IF NEW.action IN ('submit','supersede') THEN
                IF role_name IS NULL OR NOT (role_name='admin' OR (role_name='author' AND q.created_by=actor)) THEN
                    RAISE EXCEPTION 'Authoring permission required' USING ERRCODE='42501'; END IF;
            ELSE
                IF role_name IS NULL OR role_name NOT IN ('admin','reviewer') THEN
                    RAISE EXCEPTION 'Review permission required' USING ERRCODE='42501'; END IF;
            END IF;
            IF NEW.action IN ('approve','return') AND (actor=q.created_by OR actor=v.authored_by OR actor=previous.actor_user_id) THEN
                RAISE EXCEPTION 'Independent reviewer required' USING ERRCODE='42501'; END IF;
            IF NOT COALESCE((
                (NEW.action='submit' AND COALESCE(previous.state,'draft')='draft') OR
                (NEW.action IN ('approve','return','supersede') AND previous.state='in_review') OR
                (NEW.action='retire' AND previous.state='approved')
            ), false) THEN RAISE EXCEPTION 'Invalid review transition' USING ERRCODE='23514'; END IF;
            RETURN NEW;
        END $$;

CREATE FUNCTION question_public_content(c jsonb) RETURNS jsonb LANGUAGE plpgsql IMMUTABLE STRICT
        SET search_path = pg_catalog, public AS $$
        DECLARE result jsonb; values_json jsonb;
        BEGIN
            result := jsonb_build_object('kind',c->'kind','item_id',c->'item_id','prompt',c->'prompt','points',c->'points');
            CASE c->>'kind'
                WHEN 'single_choice', 'multiple_select' THEN
                    SELECT jsonb_agg(jsonb_build_object('id',x->'id','text',x->'text') ORDER BY n) INTO values_json
                        FROM jsonb_array_elements(c->'options') WITH ORDINALITY AS a(x,n);
                    result := result || jsonb_build_object('options',values_json);
                WHEN 'numeric' THEN result := result || jsonb_build_object('unit_label',c->'unit_label');
                WHEN 'short_answer', 'essay' THEN result := result || jsonb_build_object('max_response_characters',c->'max_response_characters');
                WHEN 'file_upload' THEN result := result || jsonb_build_object('allowed_extensions',c->'allowed_extensions','max_files',c->'max_files','max_bytes_per_file',c->'max_bytes_per_file');
                WHEN 'passage_group' THEN
                    SELECT jsonb_agg(public.question_public_content(x) ORDER BY n) INTO values_json FROM jsonb_array_elements(c->'children') WITH ORDINALITY AS a(x,n);
                    result := result || jsonb_build_object('passage',c->'passage','children',values_json);
                ELSE RAISE EXCEPTION 'Unknown content kind' USING ERRCODE='23514';
            END CASE;
            RETURN result;
        END $$;

CREATE FUNCTION media_asset_guard() RETURNS trigger LANGUAGE plpgsql
        SET search_path=pg_catalog,public AS $$
        BEGIN
            IF TG_OP='DELETE' THEN RAISE EXCEPTION 'Media history is retained' USING ERRCODE='42501'; END IF;
            PERFORM id FROM public.institutions WHERE id=NEW.tenant_id FOR UPDATE;
            IF TG_OP='INSERT' THEN
                IF NEW.state<>'pending' OR NEW.revision<>1 OR NEW.scanner_version IS NOT NULL OR NEW.error_code IS NOT NULL THEN
                    RAISE EXCEPTION 'New media starts pending' USING ERRCODE='23514'; END IF;
                RETURN NEW;
            END IF;
            IF NEW.revision::bigint<>OLD.revision::bigint+1 OR OLD.state NOT IN ('pending','quarantine') THEN
                RAISE EXCEPTION 'Media revision or terminal state conflict' USING ERRCODE='23514'; END IF;
            IF OLD.state='pending' THEN
                IF NEW.state<>'quarantine' OR NEW.scanner_version IS NOT NULL OR NEW.error_code IS NOT NULL OR
                    (to_jsonb(NEW)-ARRAY['state','revision','object_key','object_version','digest','byte_length','width','height'])
                    IS DISTINCT FROM (to_jsonb(OLD)-ARRAY['state','revision','object_key','object_version','digest','byte_length','width','height']) THEN
                    RAISE EXCEPTION 'Only initial byte capture may leave pending' USING ERRCODE='23514'; END IF;
            ELSE
                IF (to_jsonb(NEW)-ARRAY['state','revision','scanner_version','error_code'])
                    IS DISTINCT FROM (to_jsonb(OLD)-ARRAY['state','revision','scanner_version','error_code']) OR
                    NOT ((NEW.state IN ('clean','rejected') AND NEW.scanner_version IS NOT NULL AND NEW.error_code IS NULL)
                        OR (NEW.state='quarantine' AND NEW.scanner_version IS NULL AND NEW.error_code='scan_unavailable')) THEN
                    RAISE EXCEPTION 'Stored bytes and descriptions are immutable' USING ERRCODE='23514'; END IF;
            END IF;
            RETURN NEW;
        END $$;

CREATE TRIGGER media_asset_guard BEFORE INSERT OR UPDATE OR DELETE ON media_assets FOR EACH ROW EXECUTE FUNCTION media_asset_guard();

CREATE FUNCTION version_media_guard() RETURNS trigger LANGUAGE plpgsql
        SET search_path=pg_catalog,public AS $$
        DECLARE creation bigint; number integer; revision integer; asset public.media_assets%ROWTYPE;
        BEGIN
            IF TG_OP<>'INSERT' THEN RAISE EXCEPTION 'Version media is immutable' USING ERRCODE='42501'; END IF;
            SELECT v.created_txid,v.version_number,q.revision INTO creation,number,revision FROM public.question_versions v
                JOIN public.questions q ON q.tenant_id=v.tenant_id AND q.id=v.question_id
                WHERE v.tenant_id=NEW.tenant_id AND v.question_id=NEW.question_id AND v.id=NEW.version_id;
            IF creation IS NULL OR creation<>txid_current() OR number<>revision OR EXISTS
                (SELECT 1 FROM public.question_review_events e WHERE e.tenant_id=NEW.tenant_id AND e.version_id=NEW.version_id) THEN
                RAISE EXCEPTION 'Attachments belong to version creation only' USING ERRCODE='23514'; END IF;
            SELECT * INTO asset FROM public.media_assets WHERE tenant_id=NEW.tenant_id AND id=NEW.asset_id AND state='clean';
            IF asset.id IS NULL OR NOT EXISTS (SELECT 1 FROM public.memberships m WHERE m.tenant_id=NEW.tenant_id
                AND m.user_id=NULLIF(current_setting('app.actor_user_id',true),'')::uuid AND m.status='active'
                AND (m.role='admin' OR (m.role='author' AND asset.created_by=m.user_id))) THEN
                RAISE EXCEPTION 'Clean owned media required' USING ERRCODE='23514'; END IF;
            IF (SELECT count(*) FROM public.question_version_media l WHERE l.tenant_id=NEW.tenant_id AND l.version_id=NEW.version_id)>=16 THEN
                RAISE EXCEPTION 'Too many media attachments' USING ERRCODE='23514'; END IF;
            RETURN NEW;
        END $$;

CREATE TRIGGER version_media_guard BEFORE INSERT OR UPDATE OR DELETE ON question_version_media FOR EACH ROW EXECUTE FUNCTION version_media_guard();

CREATE FUNCTION question_import_receipt_guard() RETURNS trigger LANGUAGE plpgsql
        SET search_path=pg_catalog,public AS $$
        DECLARE item jsonb; creation bigint;
        BEGIN
            IF TG_OP<>'INSERT' THEN
                RAISE EXCEPTION 'Import receipts are immutable' USING ERRCODE='42501'; END IF;
            IF jsonb_typeof(NEW.result)<>'array' OR jsonb_array_length(NEW.result)<>NEW.item_count THEN
                RAISE EXCEPTION 'Import result count mismatch' USING ERRCODE='23514'; END IF;
            IF (SELECT count(DISTINCT x->>'reference') FROM jsonb_array_elements(NEW.result) x)<>NEW.item_count OR
               (SELECT count(DISTINCT (x->>'question_id')::uuid) FROM jsonb_array_elements(NEW.result) x)<>NEW.item_count OR
               (SELECT count(DISTINCT (x->>'version_id')::uuid) FROM jsonb_array_elements(NEW.result) x)<>NEW.item_count THEN
                RAISE EXCEPTION 'Import references must be distinct' USING ERRCODE='23514'; END IF;
            FOR item IN SELECT value FROM jsonb_array_elements(NEW.result) LOOP
                IF jsonb_typeof(item)<>'object' OR NOT (item ?& ARRAY['reference','question_id','version_id']) OR
                   item-ARRAY['reference','question_id','version_id'] <> '{}'::jsonb OR
                   jsonb_typeof(item->'reference')<>'string' OR jsonb_typeof(item->'question_id')<>'string' OR
                   jsonb_typeof(item->'version_id')<>'string' OR item->>'reference' !~ '^[a-z][a-z0-9_-]{0,63}$' THEN
                    RAISE EXCEPTION 'Invalid import receipt mapping' USING ERRCODE='23514'; END IF;
                creation := NULL;
                SELECT v.created_txid INTO creation FROM public.question_versions v JOIN public.questions q
                    ON q.tenant_id=v.tenant_id AND q.id=v.question_id
                    WHERE v.tenant_id=NEW.tenant_id AND v.question_id=(item->>'question_id')::uuid AND v.id=(item->>'version_id')::uuid
                    AND v.version_number=1 AND q.revision=1 AND v.authored_by=NEW.actor_user_id AND q.created_by=NEW.actor_user_id
                    AND NOT EXISTS (SELECT 1 FROM public.question_review_events e WHERE e.tenant_id=v.tenant_id AND e.version_id=v.id);
                IF creation IS NULL OR creation<>txid_current() THEN
                    RAISE EXCEPTION 'Receipt must bind new owned drafts in the same transaction' USING ERRCODE='23514'; END IF;
            END LOOP;
            RETURN NEW;
        END $$;

CREATE TRIGGER question_import_receipt_guard BEFORE INSERT OR UPDATE OR DELETE ON question_import_receipts FOR EACH ROW EXECUTE FUNCTION question_import_receipt_guard();
